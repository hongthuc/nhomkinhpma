    // ---------------------------------------------------------
    // 6. LOGIC BANNER SLIDER TỰ TRƯỢT + HỖ TRỢ VUỐT CẢM ỨNG
    // ---------------------------------------------------------
    let currentSlide = 0;
    const totalSlides = 2;
    let slideInterval = null;

    function updateSlideDisplay() {
      const slides = document.querySelectorAll('.hero-slide');
      slides.forEach((slide, index) => {
        if (index === currentSlide) {
          slide.classList.add('active');
          slide.classList.remove('inactive');
        } else {
          slide.classList.remove('active');
          slide.classList.add('inactive');
        }
      });

      for (let i = 0; i < totalSlides; i++) {
        const dot = document.getElementById(`dot-${i}`);
        if (dot) {
          dot.className = i === currentSlide
            ? 'w-6 h-2 rounded-full bg-pma-gold transition-all'
            : 'w-2 h-2 rounded-full bg-white/50 hover:bg-white transition-all';
        }
      }
    }

    function nextSlide() {
      currentSlide = (currentSlide + 1) % totalSlides;
      updateSlideDisplay();
      restartSlideTimer();
    }

    function prevSlide() {
      currentSlide = (currentSlide - 1 + totalSlides) % totalSlides;
      updateSlideDisplay();
      restartSlideTimer();
    }

    function goToSlide(index) {
      currentSlide = index;
      updateSlideDisplay();
      restartSlideTimer();
    }

    function startSlideTimer() {
      slideInterval = setInterval(() => {
        currentSlide = (currentSlide + 1) % totalSlides;
        updateSlideDisplay();
      }, 5500);
    }

    function restartSlideTimer() {
      clearInterval(slideInterval);
      startSlideTimer();
    }

    // Hỗ trợ vuốt cảm ứng (Swipe) trên điện thoại cho Banner
    function initSliderTouch() {
      const slider = document.getElementById('heroSlider');
      let startX = 0;
      slider.addEventListener('touchstart', (e) => {
        startX = e.touches[0].clientX;
      }, { passive: true });
      slider.addEventListener('touchend', (e) => {
        const diff = startX - e.changedTouches[0].clientX;
        if (Math.abs(diff) > 45) {
          if (diff > 0) nextSlide();
          else prevSlide();
        }
      }, { passive: true });
      slider.addEventListener('mouseenter', () => clearInterval(slideInterval));
      slider.addEventListener('mouseleave', () => startSlideTimer());
    }

    function filterProductsByCategory(category) {
      document.getElementById('searchStatusBar').classList.add('hidden');
      setActiveFilterBtn(category);
      if (category === 'Tất cả') {
        renderProducts(PRODUCTS);
      } else {
        const filtered = PRODUCTS.filter(p => p.category.toLowerCase() === category.toLowerCase());
        renderProducts(filtered);
      }

    function filterProductsByKeyword(keyword) {
      const q = keyword.trim().toLowerCase();
      if (!q) {
        resetProductFilter();
        return;
      }
      const filtered = PRODUCTS.filter(p =>
        p.name.toLowerCase().includes(q) || p.category.toLowerCase().includes(q)
      );
      renderProducts(filtered);
      const bar = document.getElementById('searchStatusBar');
      document.getElementById('searchStatusText').textContent = `Đang hiển thị ${filtered.length} sản phẩm cho từ khóa "${keyword}"`;
      bar.classList.remove('hidden');
      document.getElementById('san-pham').scrollIntoView({ behavior: 'smooth' });
    }

      document.getElementById('searchInput').value = '';
      document.getElementById('mobileSearchInput').value = '';
